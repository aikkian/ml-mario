# ----------------------------------------------------------------------------
# ml-mario dependencies
#
# These versions are PINNED on purpose. The single most common reason this kind
# of project fails to run is a version mismatch between the (older) Mario
# emulator libraries and the (newer) reinforcement-learning library. This set is
# chosen to work together.
#
# Install with:   pip install -r requirements.txt
#
# If installation fails (see README "Troubleshooting"), the documented fallback
# is the older combo: stable-baselines3==1.8.0 + gym==0.21.0
# ----------------------------------------------------------------------------

# The NES emulator and the Super Mario Bros environment (the ROM ships inside
# this package, so you do NOT need to find a ROM yourself).
gym-super-mario-bros==7.4.0
nes-py==8.2.1

# The "rules of the game" interface that RL algorithms talk to.
gym==0.26.2

# The reinforcement-learning algorithms (we use PPO). Installs PyTorch too.
stable-baselines3==2.3.2

# Bridges the old `gym` API to the new `gymnasium` API used by SB3.
shimmy>=1.0

# Image preprocessing (shrink + grayscale the screen so learning is faster).
opencv-python

# Live training charts so you can watch the agent improve.
tensorboard
